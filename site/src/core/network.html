 45def send_framed(conn: socket.socket, cipher: XorCipher, message: str) -> None:
 46    """
 47    Envoie un message avec framing : header 10 octets (taille) + payload XOR.
 48
 49    Le header est la taille du payload chiffre, encodee en ASCII sur 10 chars.
 50    """
 51    payload = cipher.process(message.encode("utf-8"))
 52    header = f"{len(payload):<10}".encode("ascii")
 53    conn.sendall(header + payload)

 56def recv_framed(conn: socket.socket, cipher: XorCipher) -> str | None:
 57    """
 58    Recoit un message avec framing : header 10 octets + payload XOR.
 59
 60    Returns:
 61        Message dechiffre ou None si la connexion est fermee.
 62    """
 63    # Lire le header
 64    header = _recv_exact(conn, HEADER_SIZE)
 65    if header is None:
 66        return None
 67
 68    try:
 69        payload_size = int(header.decode("ascii").strip())
 70    except (ValueError, UnicodeDecodeError):
 71        log.warning("Header invalide recu")
 72        return None
 73
 74    if payload_size <= 0 or payload_size > 1_000_000:
 75        log.warning("Taille de payload invalide : %d", payload_size)
 76        return None
 77
 78    # Lire le payload
 79    payload = _recv_exact(conn, payload_size)
 80    if payload is None:
 81        return None
 82
 83    return cipher.process(payload).decode("utf-8", errors="replace")

 86def _recv_exact(conn: socket.socket, size: int) -> bytes | None:
 87    """Recoit exactement `size` octets depuis la socket."""
 88    data = b""
 89    while len(data) < size:
 90        chunk = conn.recv(size - len(data))
 91        if not chunk:
 92            return None
 93        data += chunk
 94    return data

 97class PKIServer:
 98    """Serveur TCP multi-thread pour la gestion de PKI."""
 99
100    def __init__(self, host: str, port: int, xor_key: int,
101                 db, command_handler, ipv6: bool = False,
102                 tls_context=None):
103        """
104        Args:
105            host            : adresse d'ecoute.
106            port            : port d'ecoute.
107            xor_key         : cle XOR partagee avec les clients.
108            db              : instance Database.
109            command_handler : fonction(session, command) -> reponse.
110            ipv6            : True pour ecouter en IPv6.
111            tls_context     : ssl.SSLContext ou None.
112        """
113        self.host = host
114        self.port = port
115        self.xor_key = xor_key
116        self.ipv6 = ipv6
117        self.cipher = XorCipher(xor_key)
118        self.db = db
119        self.command_handler = command_handler
120        self._server_socket: socket.socket | None = None
121        self._running = False
122        self.tls_context = tls_context  # ssl.SSLContext ou None
123
124    def start(self) -> None:
125        """Demarre le serveur et accepte les connexions."""
126        af = socket.AF_INET6 if self.ipv6 else socket.AF_INET
127        self._server_socket = socket.socket(af, socket.SOCK_STREAM)
128        if self.ipv6:
129            # Accepter connexions IPv6 uniquement (pas dual-stack)
130            self._server_socket.setsockopt(socket.IPPROTO_IPV6, socket.IPV6_V6ONLY, 1)
131        self._server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
132        self._server_socket.bind((self.host, self.port))
133        self._server_socket.listen(5)
134        self._running = True
135
136        log.info("Serveur PKI demarre sur %s:%s", self.host, self.port)
137
138        try:
139            while self._running:
140                try:
141                    conn, addr = self._server_socket.accept()
142                    log.info("Connexion entrante : %s:%s", addr[0], addr[1])
143                    t = threading.Thread(
144                        target=self._handle_client,
145                        args=(conn, addr),
146                        daemon=True,
147                    )
148                    t.start()
149                except OSError:
150                    if self._running:
151                        raise
152                    break
153        except KeyboardInterrupt:
154            log.info("Arret du serveur (Ctrl+C)")
155        finally:
156            self.stop()
157
158    def stop(self) -> None:
159        """Arrete le serveur proprement."""
160        self._running = False
161        if self._server_socket:
162            self._server_socket.close()
163            self._server_socket = None
164            log.info("Serveur arrete")
165
166    def _handle_client(self, conn: socket.socket, addr: tuple) -> None:
167        """Gere une connexion client dans un thread dedie."""
168        # Wrapping TLS si active
169        if self.tls_context:
170            import ssl
171            try:
172                conn = self.tls_context.wrap_socket(conn, server_side=True)
173            except ssl.SSLError as e:
174                log.warning("Echec handshake TLS depuis %s : %s", addr[0], e)
175                conn.close()
176                return
177        cipher = XorCipher(self.xor_key)
178        session = ClientSession(conn, addr, cipher)
179
180        try:
181            # Envoi du message hello avec challenge
182            banner = (
183                "╔══════════════════════════════════════════╗\r\n"
184                "║     SAE302 PKI Manager  v1.0             ║\r\n"
185                "║  Tapez 'help' pour la liste des commandes║\r\n"
186                "╚══════════════════════════════════════════╝\r\n"
187                f"CHALL:{session.challenge}"
188            )
189            send_framed(conn, cipher, banner)
190
191            while self._running:
192                message = recv_framed(conn, cipher)
193                if message is None:
194                    log.info("Client deconnecte : %s", session)
195                    break
196
197                message = message.strip()
198                if not message:
199                    continue
200
201                log.debug("RECV [%s]: %s", session, message)
202
203                # Traitement de la commande
204                response = self.command_handler(session, message, self.db)
205
206                # Envoi de la reponse
207                send_framed(conn, cipher, response)
208
209                # Si bye, fermer la connexion
210                if message.lower() == "bye":
211                    break
212
213        except ConnectionError:
214            log.info("Connexion perdue : %s", session)
215        except Exception as e:
216            log.error("Erreur client %s : %s", session, e)
217        finally:
218            conn.close()
