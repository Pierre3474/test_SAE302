session.py — Web session management for SAE302 PKI Web Interface.

Provides WebSession and WebSessionStore for managing authenticated HTTP sessions with UUID tokens and TTL-based cleanup.

 13class WebSession:
 14    """Represents a single authenticated web session."""
 15
 16    def __init__(self, token: str, username: str, role: str):
 17        self.token = token
 18        self.username = username
 19        self.role = role
 20        self.last_activity = time.time()
 21        self.lock = threading.Lock()
 22        # Connexion TCP persistante vers le serveur PKI (evite la ré-auth TOTP a chaque requete)
 23        self._proxy = None
 24
 25    def touch(self):
 26        """Update last activity timestamp."""
 27        with self.lock:
 28            self.last_activity = time.time()
 29
 30    def is_expired(self, ttl: float) -> bool:
 31        """Return True if session has exceeded TTL seconds of inactivity."""
 32        with self.lock:
 33            return (time.time() - self.last_activity) > ttl
 34
 35    def __repr__(self) -> str:
 36        return f"WebSession(user={self.username!r}, role={self.role!r})"

 39class WebSessionStore:
 40    """
 41    Thread-safe in-memory store for WebSession objects.
 42
 43    Sessions expire after TTL seconds of inactivity.
 44    A background daemon thread handles periodic cleanup.
 45    """
 46
 47    TTL = 3600  # seconds
 48
 49    def __init__(self, ttl: int = TTL, cleanup_interval: int = 300):
 50        self._sessions: dict[str, WebSession] = {}
 51        self._lock = threading.Lock()
 52        self.ttl = ttl
 53        self._cleanup_interval = cleanup_interval
 54        self._start_cleanup_thread()
 55
 56    def _start_cleanup_thread(self):
 57        t = threading.Thread(target=self._cleanup_loop, daemon=True)
 58        t.start()
 59
 60    def _cleanup_loop(self):
 61        while True:
 62            time.sleep(self._cleanup_interval)
 63            self._purge_expired()
 64
 65    def _purge_expired(self):
 66        with self._lock:
 67            expired = [tok for tok, sess in self._sessions.items()
 68                       if sess.is_expired(self.ttl)]
 69            for tok in expired:
 70                del self._sessions[tok]
 71
 72    def create(self, username: str, role: str) -> str:
 73        """Create a new session and return the token."""
 74        token = str(uuid.uuid4())
 75        session = WebSession(token, username, role)
 76        with self._lock:
 77            self._sessions[token] = session
 78        return token
 79
 80    def get(self, token: str) -> "WebSession | None":
 81        """Return session for token, updating activity, or None if not found/expired."""
 82        with self._lock:
 83            session = self._sessions.get(token)
 84        if session is None:
 85            return None
 86        if session.is_expired(self.ttl):
 87            self.delete(token)
 88            return None
 89        session.touch()
 90        return session
 91
 92    def delete(self, token: str) -> None:
 93        """Remove a session by token."""
 94        with self._lock:
 95            self._sessions.pop(token, None)
 96
 97    def count(self) -> int:
 98        """Return number of active sessions."""
 99        with self._lock:
100            return len(self._sessions)
