 30    def connect(self) -> None:
 31        """Cree le pool de connexions a partir des variables d'environnement."""
 32        if self._pool is not None:
 33            return
 34        dsn = {
 35            "host": os.getenv("POSTGRES_HOST", "127.0.0.1"),
 36            "port": int(os.getenv("POSTGRES_PORT", "5432")),
 37            "dbname": os.getenv("POSTGRES_DB", "sae302_pki"),
 38            "user": os.getenv("POSTGRES_USER", "sae302"),
 39            "password": os.getenv("POSTGRES_PASSWORD", ""),
 40        }
 41        self._pool = pool.ThreadedConnectionPool(
 42            self.minconn, self.maxconn, **dsn
 43        )
 44        log.info("Pool PostgreSQL ouvert (%d–%d connexions)", self.minconn, self.maxconn)
 45        self._run_migrations()

 47    def _run_migrations(self) -> None:
 48        """Applique les migrations de schema manquantes (idempotent)."""
 49        with self.cursor() as cur:

 52                "recovery_codes TEXT DEFAULT NULL"
 53            )

 55    def close(self) -> None:
 56        """Ferme le pool de connexions."""
 57        if self._pool is not None:
 58            self._pool.closeall()
 59            self._pool = None
 60            log.info("Pool PostgreSQL ferme")

 62    @contextmanager
 63    def cursor(self, commit: bool = True):
 64        """Context manager qui fournit un curseur avec auto-commit/rollback."""
 65        if self._pool is None:
 66            raise RuntimeError("Le pool n'est pas initialise (appelez connect())")
 67        conn = self._pool.getconn()
 68        try:
 69            with conn.cursor(cursor_factory=extras.DictCursor) as cur:
 70                yield cur
 71                if commit:
 72                    conn.commit()
 73        except Exception:
 74            conn.rollback()
 75            raise
 76        finally:
 77            self._pool.putconn(conn)

418        import json
419        user = self.get_user_by_id(user_id)

425            return False

427        if code_upper not in codes:
428            return False
429        codes.remove(code_upper)
430        with self.cursor() as cur:
431            cur.execute(
432                "UPDATE users SET recovery_codes = %s WHERE id = %s",
433                (json.dumps(codes) if codes else None, user_id),
434            )

455    def record_failed_login(self, username: str) -> int:
456        """Incremente le compteur d'echecs et verrouille si seuil atteint.
457        Retourne le nombre de tentatives apres incrementation."""
458        with self.cursor() as cur:
459            cur.execute(
460                "UPDATE users SET failed_attempts = failed_attempts + 1, "
461                "locked_until = CASE "
462                "  WHEN failed_attempts + 1 >= %s "
463                "  THEN CURRENT_TIMESTAMP + INTERVAL '%s minutes' "
464                "  ELSE locked_until "
465                "END "
466                "WHERE username = %s "
467                "RETURNING failed_attempts",
468                (self.MAX_FAILED_ATTEMPTS, self.LOCKOUT_MINUTES, username),
469            )
470            row = cur.fetchone()
471            return row["failed_attempts"] if row else 0

481    def is_account_locked(self, username: str) -> bool:
482        """Retourne True si le compte est actuellement verrouille."""
483        with self.cursor(commit=False) as cur:
484            cur.execute(
485                "SELECT locked_until FROM users WHERE username = %s",
486                (username,),
